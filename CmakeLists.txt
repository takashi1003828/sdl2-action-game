cmake_minimum_required(VERSION 3.10)
project(actionman_game)

#C++ 17を使用
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ヘッダーファイルの検索パス
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

# homebrewのパスを検索対象に追加
list(APPEND CMAKE_PREFIX_PATH "/opt/homebrew")

# SDL2ライブラリを検索
find_package(SDL2 REQUIRED)

find_path(SDL2_TTF_INCLUDE_DIR SDL_ttf.h
    HINTS /opt/homebrew/include/SDL2 /usr/local/include/SDL2
)
find_library(SDL2_TTF_LIBRARY NAMES SDL2_ttf
    HINTS /opt/homebrew/lib /usr/local/lib
)

#コンパイルするソースファイルを指定
set(SOURCES
    src/main.cpp
    src/InputManager.cpp
    src/Game.cpp
    src/Player.cpp
    src/IdleState.cpp
    src/WalkState.cpp
    src/JumpState.cpp
    src/DashState.cpp
    src/AirState.cpp
    src/Camera.cpp
    src/Map.cpp
    src/CollisionManager.cpp
)

# 実行ファイルを作成
add_executable(actionman_game ${SOURCES})

# ヘッダーファイルの場所を指定
target_include_directories(actionman_game PRIVATE ${SDL2_INCLUDE_DIRS})

# ライブラリ本体の場所とライブラリ名を指定
target_link_libraries(actionman_game PRIVATE ${SDL2_LIBRARIES})

target_include_directories(actionman_game PRIVATE ${SDL2_TTF_INCLUDE_DIR})
target_link_libraries(actionman_game PRIVATE ${SDL2_TTF_LIBRARY})
